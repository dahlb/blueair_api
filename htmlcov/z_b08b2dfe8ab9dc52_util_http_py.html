18async def send_with_timeout(request: Awaitable[ClientResponse]) -> ClientResponse: 

19 """Await an aiohttp request, converting a timeout into a package error. 

20 

21 The request itself should be made with ``timeout=REQUEST_TIMEOUT``. 

22 aiohttp reports an expired total timeout as a plain ``TimeoutError``, 

23 which is not a ``ClientError``; it is re-raised as ``RequestTimeoutError`` 

24 so callers that retry client errors retry timeouts too. 

25 

26 Args: 

27 request: The pending ``ClientSession.get`` or ``post`` call. 

28 

29 Returns: 

30 The HTTP response. 

31 

32 Raises: 

33 RequestTimeoutError: If the request does not complete in time. 

34 """ 

35 try: 

36 return await request 

37 except TimeoutError as error: 

38 _LOGGER.debug("Blueair request timed out after %s seconds", REQUEST_TIMEOUT_SECONDS) 

39 raise RequestTimeoutError( 

40 f"Blueair request timed out after {REQUEST_TIMEOUT_SECONDS} seconds" 

41 ) from error

44def request_with_logging[**P]( 

45 func: Callable[P, Awaitable[ClientResponse]], 

46) -> Callable[P, Awaitable[ClientResponse]]: 

47 """Wrap an async HTTP call, logging its request and response for debugging. 

48 

49 Request headers, params and JSON bodies and the response headers and JSON 

50 body pass through ``clean_dictionary_for_logging``, so tokens and 

51 credentials are masked. 

52 

53 Args: 

54 func: The HTTP call to wrap. It must take ``url`` as a keyword argument. 

55 

56 Returns: 

57 The wrapped call, with the same signature, name and docstring. 

58 """ 

59 

60 @functools.wraps(func) 

61 async def request_with_logging_wrapper(*args: P.args, **kwargs: P.kwargs) -> ClientResponse: 

62 """Log the outgoing request, call ``func``, then log the response. 

63 

64 Args: 

65 args: Positional arguments for ``func``. 

66 kwargs: Keyword arguments for ``func``, including ``url``. 

67 

68 Returns: 

69 The response returned by ``func``. 

70 """ 

71 url = kwargs["url"] 

72 request_message = f"sending {url} request" 

73 headers = kwargs.get("headers") 

74 if headers is not None: 

75 request_message += f" headers: {clean_dictionary_for_logging(headers)}" 

76 params = kwargs.get("params") 

77 if params is not None: 

78 request_message += f" params: {clean_dictionary_for_logging(params)}" 

79 json_body = kwargs.get("json_body") 

80 if json_body is not None: 

81 request_message += f" json: {clean_dictionary_for_logging(json_body)}" 

82 _LOGGER.debug(request_message) 

83 response = await func(*args, **kwargs) 

84 _LOGGER.debug("response headers:%s", clean_dictionary_for_logging(response.headers)) 

85 _LOGGER.debug("response status: %s", response.status) 

86 try: 

87 response_json = await response.json(content_type=None) 

88 _LOGGER.debug("response json: %s", clean_dictionary_for_logging(response_json)) 

89 except ValueError, UnicodeDecodeError: 

90 response_text = await response.text() 

91 _LOGGER.debug("response raw: %s", response_text) 

92 return response 

93 

94 return request_with_logging_wrapper
